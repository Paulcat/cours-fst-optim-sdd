\begin{itemize}[label=\tiny$\blacksquare$]
	\item Sur cette représentation simple:
	\begin{itemize}[label=-]
		\item à chaque \alert{connection} ${i}$ est affecté un \alert{poids} multiplicatif $w_i$
		\item un neurone de sortie \alert{additionne} les entrées
	\end{itemize}
	
	\pause
	
	\item On note \alert{$\theta \eqdef (w_1,w_2)$} l'ensemble des paramètres du graphe. 
	
	Ici on a représenté la fonction
	\eq{\tcbhighmath{
		y = f_\theta(x) = \pause w_1 x_1 + w_2 x_2
	}}
\end{itemize}

	\item Critère d'optimisation: à partir de \alert{$n$ données labellisées} $(x^{(i)},y^{(i)}) \in \RR^d \times \Yy$
	\eq{
		\min_{\theta \in \RR^d} \sum_{i=1}^n \ell({\theta^\top x^{(i)}},y^{(i)})
	}
	où $\ell$ est une fonction perte adéquate.
	
	\pause
	
	\begin{tcolorbox}[bottom=0pt]
	\item \alert{Remarque}: $\theta \mapsto f_\theta(x)$ est linéaire, mais pas nécessairement $x\mapsto f_\theta(x)$:

		\item \alert{Méthodes à noyau}: remplacer $x$ par $\phi(x) \in \RR^D$, avec $D \gg d$
		\item \alert{Apprentissage profond}: introduire de la non-linéarité pour apprendre $\phi$
	\end{itemize}
	\end{tcolorbox}
\end{itemize}

\begin{frame}
\frametitle{Calcul du gradient}

\begin{itemize}[label=\tiny$\blacksquare$]
	\item La minimisation de $J(\theta)$ nécessite de pouvoir dériver la sortie du réseau par rapport aux paramètres $\theta = (\theta_1,\ldots,\theta_L)$ du réseau.
	
	\item Autrement dit, on doit calculer
	\eq{
		\nabla J = \left( \frac{\partial J}{\partial \theta_1}, \ldots,  \frac{\partial J}{\partial \theta_L} \right)^\top
	}
	
	\item Pour estimer ce gradient, on exploite le principe de la \alert{rétropropagation du gradient} ("backpropagation")
\end{itemize}

		\item $\bm{g} \leftarrow W_l^\top \bm{g}$
	\end{itemize}
	$\bullet$ Renvoyer ${\nabla_W J}$
	\end{tcolorbox}
	\textcolor{mLightBrown}{Rétropropagation}
	\end{column}
	\end{columns}
	
	
	\item Plus généralement: \alert{différentiation automatique}
\end{itemize}


\end{frame}
