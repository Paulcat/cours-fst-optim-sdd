\section{Optimisation}

\subsection{Formulation d'un problème d'optimisation}
On souhaite déterminer un parallélépipède rectangle de volume unité $(V=1)$ avec une surface $S$ minimale.
\begin{enumerate}
	\item Formuler ce problème comme un problème d'optimisation en précisant les variables du problème, le critère et les contraintes.
	\item Transformer l'expression du critère pour faire apparaître une fonction de deux variables. Que devient le nouveau problème d'optimisation?
	\item Représenter le critère dans le plan d’une seule variable, en fixant une valeur pour l’autre variable. Donner sur le graphe les valeurs du critère pour quelques valeurs de la variable libre.
\end{enumerate}

\subsection{Méthode de Newton}
Soit un problème d'optimisation à deux variables $x$ et $y$ dont le critère (convexe) est donné par l'expression suivante:
\eq{
	f(x,y) = \frac{1}{2}x^2 + \frac{9}{2}y^2 -x - 18y +20.
}

\begin{enumerate}
	\item Expliquer pourquoi la convexité est une propriété importante en optimisation.
	\item Donner l'expression du gradient $\mathbf{J}(x,y)$ du critère et calculer le.
	\item Donner l'expression de la matrice Hessienne $\mathbf{H}(x,y)$ du critère et calculer la.
	\item Que vaut le minimiseur? Que vaut le minimum du critère?
	\item En considérant que l'algorithme de Newton est initialisé à des valeurs quelconques $x_0$ et $y_0$, dérouler cet algorithme pour déterminer les valeurs suivantes notées $x_1$ et $y_1$ en expliquant vos calculs.
	\item Que constatez-vous? Commenter.
\end{enumerate}

\section{Décompositions matricielles}
Soit $\X$ une matrice réelle de dimensions $m \times n$ avec $m > n$. On définit une matrice $\C = \X\X^\top$.
\begin{enumerate}
	\item Les valeurs propres $\lambda_i$ de la matrice $\C$ correspondent-elles aux:
	\begin{enumerate}
		\item valeurs singulières $\sigma_i$ de $\X$?
		\item valeurs singulières de $\C$?
		\item carrés des valeurs singulières $\sigma_i$ de $\X$?
	\end{enumerate}
	\item Les vecteurs propres de la matrice $\C$ correspondent-ils aux:
	\begin{enumerate}
		\item vecteurs propres de $\X$?
		\item vecteurs singuliers droits de $\X$?
		\item vecteurs singuliers gauches de $\X$?
	\end{enumerate}
\end{enumerate}
Considérons la matrice $\X$ suivante:
\eq{
	\X = \begin{bmatrix} 0 & 1\\ 1 & 2\\ 2 & 0\end{bmatrix}.
}

\begin{enumerate}
\setcounter{enumi}{2}
	\item Calculer le produit $\C = \X\X^\top$.
	\item Quels vecteurs parmi les suivants sont des vecteurs propres de la matrice $\C$? (Rappel: les vecteurs et valeurs propres $(\lambda,\xx$) sont des solutions de $\C\xx = \la\xx$)
	
	\begin{inlinelist}
		\item $\begin{bmatrix} 0 \\ 0 \\ 2\end{bmatrix}$
		\item $\begin{bmatrix} 1 \\ 0 \\ -1\end{bmatrix}$
		\item $\begin{bmatrix} 2 \\ -1 \\ \frac{1}{2}\end{bmatrix}$
		\item $\begin{bmatrix} 3 \\ 3 \\ 3\end{bmatrix}$
		\item $\begin{bmatrix} 1 \\ 3 \\ 2\end{bmatrix}$
		\item $\begin{bmatrix} 1 \\ 1 \\ -2\end{bmatrix}$
	\end{inlinelist}
	\item Quelles sont les valeurs propres associées? Combien y a-t-il de valeurs propres strictement positives?
	\item Quel est le rang de la matrice $\C$?
	\item Trouver les valeurs singulières $\sigma_1 \geq \sigma_2 \geq \sigma_3$ de $\X$.
	\item Calculer l'approximation $\X_1 = \sigma_1 \uu_1\vv_1^\top$, en utilisant les propriétés suivantes:
	\begin{itemize}[label=-]
		\item les vecteurs singuliers gauches doivent être normalisés ($\norm{\uu_k}_2 = 1$);
		\item les vecteurs singuliers droits peuvent être obtenus par $\vv_k = \frac{1}{\sigma_k}\X^\top \uu_k$ à partir des vecteurs singuliers gauches $\uu_k$ (normalisés) et des valeurs singulières $\sigma_k$. 
	\end{itemize}
	\item Calculer l'erreur d'approximation $\norm{X-X_1}_F$, où $\norm{Y}_F$ est la norme de Frobenius:
	\eq{
		\norm{Y}_F^2 = \sum_{i,j}(Y(i,j))^2.
	}
\end{enumerate}
