\begin{frame}
\frametitle{Méthodes itératives}

\begin{itemize}[label=\tiny$\blacksquare$]
	\item On veut résoudre un problème d'optimisation (continue) sans contraintes
	\eq{
		\min_{x \in \RR^n} f(x) 
	}
	
	Dans la très grande majorité des cas, il n'y a \alert{pas de solution analytique} comme pour les moindres carrés
	
	\item \textbf{Stratégie}: \alert{méthodes itératives} (algorithmes)
	\begin{itemize}
		\item[1.] on \alert{choisit} un (ou plusieurs) point initial $x_0 \in \RR^n$
		\item[2.] à chaque itération $k$, on obtient un nouveau point $x_k$ suivant un mise à jour construite à partir de la fonction $f$
		\item[3.] on réitère jusqu'à ce qu'un critère d'arrêt soit satisfait
	\end{itemize}
\end{itemize}

\end{frame}
